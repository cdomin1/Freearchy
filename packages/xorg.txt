# X11 session stack for vellum-wm (shelved Wayland path is in hyprland.txt)
xorg-server
xinit
xauth
xmodmap                   # REQUIRED: penrose parses keybinds via xmodmap
xrandr
setxkbmap
rofi                      # launcher (Super+Space)
alacritty                 # terminal (foot is wayland-only)
xset                      # DPMS / screen blanking (xinitrc)
xsetroot                  # root cursor + solid-colour fallback (xinitrc)
feh                       # wallpaper setter
ImageMagick7              # `magick` converts theme wallpapers (vellum theme)
picom                     # compositor — v12+ has native animations
dejavu                    # baseline fonts for polybar/rofi/alacritty
noto-basic?               # wider glyph coverage
polybar                   # panel (EWMH modules + FreeBSD script shims)
xfce4-notifyd             # notifications (dunst isn't ported)
materia-gtk-theme         # GTK theme (adw-gtk3-alike dark, thunar etc.)
papirus-icon-theme        # icons — Omarchy uses Papirus too
adwaita-icon-theme        # fallback icons + Adwaita cursor
gtk-murrine-engine        # GTK2 engine some themes still need
slock                     # screen lock (Super+Esc)
maim                      # screenshots (Print)
